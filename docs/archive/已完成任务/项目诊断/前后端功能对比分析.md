### 示例 3: 批量任务操作

**位置**: `client/src/components/projects/TaskList.tsx`

```tsx
// 添加状态
const [selectedTasks, setSelectedTasks] = useState<Set<string>>(new Set());
const [showBatchActions, setShowBatchActions] = useState(false);

// 批量操作处理
const handleBatchUpdateStatus = async (newStatus: string) => {
  try {
    await projectService.batchUpdateTaskStatus(
      Array.from(selectedTasks),
      newStatus
    );
    // 刷新任务列表
    await loadTasks();
    setSelectedTasks(new Set());
    setShowBatchActions(false);
  } catch (error) {
    console.error('批量更新失败:', error);
  }
};

// UI 组件
{selectedTasks.size > 0 && (
  <div className="bg-blue-50 border border-blue-200 rounded-lg p-4 mb-4">
    <div className="flex items-center justify-between">
      <span className="text-sm text-gray-700">
        已选择 {selectedTasks.size} 个任务
      </span>
      <div className="flex space-x-2">
        <Button onClick={() => handleBatchUpdateStatus('in_progress')}>
          批量开始
        </Button>
        <Button onClick={() => handleBatchUpdateStatus('completed')}>
          批量完成
        </Button>
        <Button variant="outline" onClick={() => setSelectedTasks(new Set())}>
          取消选择
        </Button>
      </div>
    </div>
  </div>
)}

<div className="space-y-2">
  {tasks.map(task => (
    <div key={task.id} className="flex items-center space-x-3">
      <input
        type="checkbox"
        checked={selectedTasks.has(task.id)}
        onChange={(e) => {
          const newSelected = new Set(selectedTasks);
          if (e.target.checked) {
            newSelected.add(task.id);
          } else {
            newSelected.delete(task.id);
          }
          setSelectedTasks(newSelected);
        }}
      />
      <TaskCard task={task} />
    </div>
  ))}
</div>
```
